cmake_minimum_required(VERSION 3.25)
project(Aulamix250 VERSION 0.4.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
if(NOT VST3_SDK_ROOT)
 message(FATAL_ERROR "Set VST3_SDK_ROOT to Steinberg VST3 SDK v3.8.0_build_66")
endif()
add_subdirectory(${VST3_SDK_ROOT} ${CMAKE_BINARY_DIR}/sdk)
if(MINGW)
 # SDK dataexchange uses std::aligned_alloc, unavailable in MinGW.
 # This plugin does not use the optional data exchange helper.
 get_target_property(sdk_sources sdk SOURCES)
 list(FILTER sdk_sources EXCLUDE REGEX "utility/dataexchange[.]cpp$")
 set_property(TARGET sdk PROPERTY SOURCES "${sdk_sources}")
endif()
smtg_add_vst3plugin(Aulamix250 PACKAGE_NAME "EMT-250 Reverb" SOURCES_LIST native/Plugin.cpp ${VST3_SDK_ROOT}/public.sdk/source/vst/vstsinglecomponenteffect.cpp)
if(WIN32)
 enable_language(RC)
 configure_file(native/UiAssets.rc.in ${CMAKE_BINARY_DIR}/UiAssets.rc @ONLY)
 target_sources(Aulamix250 PRIVATE ${CMAKE_BINARY_DIR}/UiAssets.rc)
 target_sources(Aulamix250 PRIVATE ${VST3_SDK_ROOT}/public.sdk/source/main/dllmain.cpp)
 target_link_libraries(Aulamix250 PRIVATE gdi32 user32 comdlg32 gdiplus ole32)
elseif(APPLE)
 target_sources(Aulamix250 PRIVATE ${VST3_SDK_ROOT}/public.sdk/source/main/macmain.cpp)
else()
 target_sources(Aulamix250 PRIVATE ${VST3_SDK_ROOT}/public.sdk/source/main/linuxmain.cpp)
endif()
target_link_libraries(Aulamix250 PRIVATE sdk)
target_include_directories(Aulamix250 PRIVATE ${VST3_SDK_ROOT})
enable_testing()
add_executable(dsp-test native/DspTest.cpp)
add_test(NAME DSP COMMAND dsp-test)
if(WIN32)
 add_library(Aulamix250Vst2 MODULE native/Vst2Plugin.cpp)
 target_sources(Aulamix250Vst2 PRIVATE ${CMAKE_BINARY_DIR}/UiAssets.rc)
 target_link_libraries(Aulamix250Vst2 PRIVATE sdk gdi32 user32 comdlg32 gdiplus ole32)
 set_target_properties(Aulamix250Vst2 PROPERTIES PREFIX "" OUTPUT_NAME "EMT-250 Reverb" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/VST2" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/VST2")
 add_executable(windows-host-test native/WindowsHostTest.cpp ${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/parameterchanges.cpp)
 target_link_libraries(windows-host-test PRIVATE sdk gdi32 user32 ole32)
 add_dependencies(windows-host-test Aulamix250)
 add_test(NAME WindowsHost COMMAND windows-host-test $<TARGET_FILE:Aulamix250>)
 add_executable(vst2-host-test native/Vst2HostTest.cpp)
 target_link_libraries(vst2-host-test PRIVATE user32)
 add_dependencies(vst2-host-test Aulamix250Vst2)
 add_test(NAME VST2Host COMMAND vst2-host-test $<TARGET_FILE:Aulamix250Vst2>)
endif()
